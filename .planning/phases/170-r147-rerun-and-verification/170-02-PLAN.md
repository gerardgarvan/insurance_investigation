    p163_pass <- 0L

    p163_chk <- function(label, ok) {
      if (isTRUE(ok)) {
        message(glue("  PASS  {label}"))
        p163_pass <<- p163_pass + 1L
        passed    <<- passed    + 1L
      } else {
        message(glue("  FAIL  {label}"))
        p163_fail <<- p163_fail + 1L
        failed    <<- failed    + 1L
      }

    message("\n--- Section 15as: Phase 163 DIAGNOSIS exclusion ---")

    r147_src <- if (file.exists("R/147_surveillance_modality_frequency.R")) {
      tryCatch(readLines("R/147_surveillance_modality_frequency.R"), error = function(e) character(0))
    } else character(0)

    # [1] EXCLUDED_CDM_TABLES defined in R/147
    p163_chk(
      "SMOKE-163-01 [1/4] EXCLUDED_CDM_TABLES defined in R/147",
      any(grepl("EXCLUDED_CDM_TABLES\\s*<-", r147_src))
    )

    # [2] stopifnot guard referencing EXCLUDED_CDM_TABLES present in R/147
    p163_chk(
      "SMOKE-163-01 [2/4] R/147 has stopifnot() guarding the EXCLUDED_CDM_TABLES partition",
      any(grepl("stopifnot", r147_src) & grepl("EXCLUDED_CDM_TABLES", r147_src))
    )

    # Output-gated checks [3]-[4]: latest INTERNAL workbook
    wb_163_dir  <- CONFIG$cache$outputs_dir

    } else character(0)

        a_cp <- tryCatch(openxlsx::readWorkbook(wb_163_path, sheet = "A_code_presence"),

          any(grepl("^DIAGNOSIS$", as.character(col), ignore.case = TRUE), na.rm = TRUE),

      }

          !has_z
        )
      } else {
        message("  NOTE  SMOKE-163-01 [4/4] Codeset_summary sheet absent — skipped (offline)")
        p163_pass <- p163_pass + 1L; passed <- passed + 1L
