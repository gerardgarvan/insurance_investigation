    ```r
    message("\n=== Phase 143 D-04: Fill-rate parity check ===")
    g90  <- readr::read_csv(file.path(CONFIG$output_dir, "gantt_episodes.csv"),     show_col_types = FALSE, col_types = cols(.default = col_character()))
    g180 <- readr::read_csv(file.path(CONFIG$output_dir, "gantt_episodes_180.csv"), show_col_types = FALSE, col_types = cols(.default = col_character()))
    enrich_cols <- c("drug_group","code_type","source_table",
                     "episode_dx_codes","episode_dx_categories","episode_dx_7day_confirmed")
    fillrate <- function(d, c) mean(!is.na(d[[c]]) & trimws(d[[c]]) != "")
    cmp <- data.frame(
      column   = enrich_cols,
      fill_90  = round(100 * vapply(enrich_cols, function(c) fillrate(g90,  c), numeric(1)), 1),
      fill_180 = round(100 * vapply(enrich_cols, function(c) fillrate(g180, c), numeric(1)), 1)
    )
    cmp$delta <- cmp$fill_180 - cmp$fill_90
    print(cmp)

    message("Fill-rate parity check: PASS")
    ```
