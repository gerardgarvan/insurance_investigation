```r
# ICD system split (computed at source time from map):
.doi_keys_icd9  <- grep("^[0-9]", names(DOI_CODE_MAP), value = TRUE)
.doi_keys_icd10 <- grep("^[0-9]", names(DOI_CODE_MAP), value = TRUE, invert = TRUE)

is_doi_code <- function(dx, dx_type) {
  if (length(dx) == 0) return(logical(0))
  result <- rep(FALSE, length(dx))
  valid <- !is.na(dx) & !is.na(dx_type) & dx_type %in% c("09", "10")
  if (!any(valid)) return(result)
  dx_clean <- normalize_icd(dx)
  p4 <- substr(dx_clean, 1, 4)
  p3 <- substr(dx_clean, 1, 3)
  is10 <- valid & dx_type == "10"
  is09 <- valid & dx_type == "09"
  result[is10] <- (p4[is10] %in% .doi_keys_icd10) | (p3[is10] %in% .doi_keys_icd10)
  result[is09] <- (p4[is09] %in% .doi_keys_icd9)  | (p3[is09] %in% .doi_keys_icd9)
  result
}
```

    ```r
    #' Detect whether diagnosis codes are autoimmune condition codes (DX_TYPE-gated)
    #'
    #' Prefix-matches against AUTOIMMUNE_DX_CODES from R/00_config.R, gated on DX_TYPE
    #' identically to is_doi_code(). Numeric ICD-9 keys only match ICD-9 records;
    #' alpha-leading ICD-10 keys only match ICD-10 records.
    #'
    #' @param dx      Character vector of diagnosis codes (dotted or undotted)
    #' @param dx_type Character vector of DX_TYPE values ("09" ICD-9, "10" ICD-10;
    #'   NA / "SM" / anything else -> FALSE)
    #' @return Logical vector; TRUE only when code prefix is in AUTOIMMUNE_DX_CODES
    #'   AND dx_type matches the coding system that key belongs to.
    #'
    #' @examples
    #' is_autoimmune_code(c("M05.9", "7140", "C81.9"), c("10", "09", "10"))
    #' # Returns: c(TRUE, TRUE, FALSE)
    is_autoimmune_code <- function(dx, dx_type) {
      if (length(dx) == 0) return(logical(0))
      result <- rep(FALSE, length(dx))

      valid <- !is.na(dx) & !is.na(dx_type) & dx_type %in% c("09", "10")
      if (!any(valid)) return(result)

      dx_clean <- normalize_icd(dx)          # reuse: uppercase + strip dots
      p4 <- substr(dx_clean, 1, 4)
      p3 <- substr(dx_clean, 1, 3)

      is10 <- valid & dx_type == "10"
      is09 <- valid & dx_type == "09"

      # ICD-10: match alpha-leading keys only (4-char before 3-char)
      result[is10] <- (p4[is10] %in% .autoimmune_keys_icd10) |
                      (p3[is10] %in% .autoimmune_keys_icd10)

      # ICD-9: match numeric keys only (4-char before 3-char)
      result[is09] <- (p4[is09] %in% .autoimmune_keys_icd9) |
                      (p3[is09] %in% .autoimmune_keys_icd9)

      result
    }
    ```

    ```r
    # non-NA and > SENTINEL_CUTOFF, else no valid date for that record.
    # Patients with qualifying code but no valid date: has_autoimmune_dx = FALSE, date = NA,
    # counted in QC (D-07a).
    #
    # DuckDB convention (CLAUDE.md): open, query, close in one block. Patient key is ID.
    # Never build an IN (...) list of IDs — query columns and join in R.
    # ==============================================================================
    SENTINEL_CUTOFF_AUTOIMMUNE <- as.Date("1910-01-01")   # mirrors R/47, R/48 convention

    build_autoimmune_dx_flags <- function(cohort_ids) {
      message("\n--- Building autoimmune diagnosis flags (Phase 192.1) ---")

      open_pcornet_con()
      on.exit(try(close_pcornet_con(), silent = TRUE))   # safety close if error

      # Pull DIAGNOSIS for all patients (join in R, not IN list per CLAUDE.md)
      dx_raw <- get_pcornet_table("DIAGNOSIS") %>%
        select(ID, DX, DX_TYPE, DX_DATE, ADMIT_DATE) %>%
        collect()

      close_pcornet_con()

      n_total_dx <- nrow(dx_raw)
      message(glue("  DIAGNOSIS rows loaded: {format(n_total_dx, big.mark = ',')}"))

      # Restrict to cohort patients
      dx_cohort <- dx_raw %>%
        filter(ID %in% cohort_ids)

      message(glue("  DIAGNOSIS rows for cohort: {format(nrow(dx_cohort), big.mark = ',')}"))

      # D-04: Gate on DX_TYPE; trim whitespace defensively
      dx_cohort <- dx_cohort %>%
        mutate(DX_TYPE = stringr::str_trim(DX_TYPE))

      n_wrong_type <- sum(!is.na(dx_cohort$DX_TYPE) &
                          !dx_cohort$DX_TYPE %in% c("09", "10"))
      message(glue("  DX_TYPE values excluded (not 09/10): {n_wrong_type}"))

      # Apply is_autoimmune_code() — DX_TYPE-gated prefix match
      dx_cohort <- dx_cohort %>%
        mutate(is_aim = is_autoimmune_code(DX, DX_TYPE))

      dx_aim <- dx_cohort %>% filter(is_aim)
      message(glue("  Qualifying autoimmune diagnosis records: {format(nrow(dx_aim), big.mark = ',')}"))

      # D-07: date priority — DX_DATE if valid (non-NA, post-sentinel), else ADMIT_DATE if valid
      dx_aim <- dx_aim %>%
        mutate(
          DX_DATE    = as.Date(DX_DATE),
          ADMIT_DATE = as.Date(ADMIT_DATE),
          valid_date = dplyr::case_when(
            !is.na(DX_DATE)    & DX_DATE    > SENTINEL_CUTOFF_AUTOIMMUNE ~ DX_DATE,
            !is.na(ADMIT_DATE) & ADMIT_DATE > SENTINEL_CUTOFF_AUTOIMMUNE ~ ADMIT_DATE,
            TRUE ~ as.Date(NA)
          )
        )

      # D-07a: count "qualifying code, no valid date" patients
      no_date_ids <- dx_aim %>%
        filter(is.na(valid_date)) %>%
        distinct(ID)
      message(glue("  Patients: qualifying code but no valid date (D-07a QC): {nrow(no_date_ids)}"))

      # Aggregate to one row per patient: earliest valid date
      aim_flags <- dx_aim %>%
        group_by(ID) %>%
        summarise(
          first_autoimmune_dx_date = suppressWarnings(min(valid_date, na.rm = TRUE)),
          .groups = "drop"
        ) %>%
        mutate(
          first_autoimmune_dx_date = dplyr::if_else(
            is.finite(first_autoimmune_dx_date),
            first_autoimmune_dx_date,
            as.Date(NA)
          )
        )

      # D-05b: ensure every cohort patient is represented (one row per ID)
      all_patients <- tibble::tibble(ID = cohort_ids)
      aim_flags_full <- all_patients %>%
        dplyr::left_join(aim_flags, by = "ID") %>%
        mutate(
          has_autoimmune_dx = !is.na(first_autoimmune_dx_date)
        )

      # D-05b assertion: one row per ID
      stopifnot(
        "build_autoimmune_dx_flags: nrow != n_distinct(ID)" =
          nrow(aim_flags_full) == dplyr::n_distinct(aim_flags_full$ID)
      )

      # D-07a assertion: no TRUE with NA date, no FALSE with a date
      stopifnot(
        "INVARIANT: has_autoimmune_dx TRUE but date is NA" =
          !any(aim_flags_full$has_autoimmune_dx & is.na(aim_flags_full$first_autoimmune_dx_date))
      )
      stopifnot(
        "INVARIANT: has_autoimmune_dx FALSE but date is non-NA" =
          !any(!aim_flags_full$has_autoimmune_dx & !is.na(aim_flags_full$first_autoimmune_dx_date))
      )

      n_true <- sum(aim_flags_full$has_autoimmune_dx)
      message(glue("  Patients with has_autoimmune_dx = TRUE: {n_true}"))
      message(glue("  Patients with has_autoimmune_dx = FALSE: {nrow(aim_flags_full) - n_true}"))

      aim_flags_full
    }
    ```

    Note: `min(valid_date, na.rm = TRUE)` on an all-NA group returns `Inf` (for Date class);
