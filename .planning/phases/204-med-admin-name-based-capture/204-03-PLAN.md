message("\nSection 15be: Phase 204 — MED_ADMIN Name-Based Capture (SMOKE-204-01)")

p204_pass <- 0L; p204_fail <- 0L; p204_skip <- 0L

p204_chk <- function(label, cond) {
  if (isTRUE(cond))       { message("  PASS: ", label); p204_pass <<- p204_pass + 1L
  } else if (is.na(cond)) { message("  SKIP: ", label); p204_skip <<- p204_skip + 1L
  } else                  { message("  FAIL: ", label); p204_fail <<- p204_fail + 1L }
}

ut_path <- file.path("R", "utils", "utils_treatment.R")
r26_path <- file.path("R", "26_treatment_episodes.R")

r26_lines <- if (file.exists(r26_path)) readLines(r26_path, warn = FALSE) else character(0)

# [a] return_name_capture parameter
p204_chk("SMOKE-204-01 [a] return_name_capture parameter in utils_treatment.R",
  any(grepl("return_name_capture", ut_lines, fixed = TRUE)))

# [b] MED_ADMIN_NAME source tag
p204_chk("SMOKE-204-01 [b] 'MED_ADMIN_NAME' source tag in utils_treatment.R",
  any(grepl("MED_ADMIN_NAME", ut_lines, fixed = TRUE)))

# [c] MCAP-05 safety gate stop()
p204_chk("SMOKE-204-01 [c] MCAP-05 safety gate stop() in utils_treatment.R",
  any(grepl("MCAP-05", ut_lines, fixed = TRUE)) &&
  any(grepl("stop(", ut_lines, fixed = TRUE)))

# [d] NAME: synthetic triggering_code
p204_chk("SMOKE-204-01 [d] 'NAME:' triggering_code synthesis in utils_treatment.R",
  any(grepl("NAME:", ut_lines, fixed = TRUE)))

# [e] pre-204 snapshot wired in R/26
p204_chk("SMOKE-204-01 [e] pre_p204 snapshot save in R/26_treatment_episodes.R",
  any(grepl("pre_p204", r26_lines, fixed = TRUE)))

# [f] post-204 snapshot wired in R/26
p204_chk("SMOKE-204-01 [f] post_p204 snapshot save in R/26_treatment_episodes.R",
  any(grepl("post_p204", r26_lines, fixed = TRUE)))

# [g] QC table write wired in R/26
p204_chk("SMOKE-204-01 [g] p204_name_qc save in R/26_treatment_episodes.R",
  any(grepl("p204_name_qc", r26_lines, fixed = TRUE)))

# [h] regression check block with stop() on mismatch
p204_chk("SMOKE-204-01 [h] MCAP-03 regression check with stop() in R/26",
  any(grepl("MCAP-03", r26_lines, fixed = TRUE)) &&
  any(grepl("REGRESSION FAIL", r26_lines, fixed = TRUE)))

    file.exists(pre_rds))

  p204_chk("SMOKE-204-01 [j] post-204 snapshot RDS exists (SKIP — local run)", NA)

  p204_chk("SMOKE-204-01 [j] post-204 snapshot RDS exists",
    file.exists(post_rds))
}

# [k] QC RDS file exists (HiPerGator only)
qc_rds <- file.path(CONFIG$output_dir, "internal", "p204_name_qc.rds")

} else {
  p204_chk("SMOKE-204-01 [k] p204_name_qc.rds exists",
    file.exists(qc_rds))
}

message(glue("\nSMOKE-204-01: {p204_pass} PASS / {p204_fail} FAIL / {p204_skip} SKIP"))
