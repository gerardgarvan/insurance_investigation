## Year-1 window SOURCE at patient level

The year-1 window for each patient is defined in py_raw as `year_index == max(year_index)`
per patient (the existing R/181 prior-year definition). The year window bounds are
`year_start` and `year_end` columns in py_raw.

Compute patient-level SOURCE_analytic for the year-1 window (D-07 / D-08):

```r
# Get year-1 window bounds per patient
year1_bounds <- py_raw |>
  dplyr::group_by(ID) |>
  dplyr::filter(year_index == max(year_index)) |>
  dplyr::ungroup() |>
  dplyr::select(ID, year_start, year_end)

# Encounters in year-1 window
enc_year1 <- enc_source_181_assigned |>
  dplyr::inner_join(year1_bounds, by = "ID") |>
  dplyr::filter(ADMIT_DATE >= year_start, ADMIT_DATE <= year_end)

# Modal SOURCE_analytic in year-1 window (D-07: modal, ties by most recent ADMIT_DATE, then alpha)
compute_modal_source <- function(enc_win, group_col = "ID",
                                  source_col = "SOURCE_analytic", date_col = "ADMIT_DATE") {
  enc_win |>
    dplyr::group_by(dplyr::across(dplyr::all_of(group_col)),
                    dplyr::across(dplyr::all_of(source_col))) |>
    dplyr::summarise(
      n         = dplyr::n(),
      last_date = max(.data[[date_col]]),
      .groups   = "drop"
    ) |>
    dplyr::group_by(dplyr::across(dplyr::all_of(group_col))) |>
    dplyr::slice_max(order_by = n, n = 1L, with_ties = TRUE) |>
    dplyr::arrange(dplyr::across(dplyr::all_of(group_col)),
                   dplyr::desc(last_date),
                   dplyr::across(dplyr::all_of(source_col))) |>
    dplyr::slice_head(n = 1L) |>
    dplyr::ungroup() |>
    dplyr::rename(modal_source = dplyr::all_of(source_col)) |>
    dplyr::select(dplyr::all_of(group_col), modal_source)
}

pat_source_year1    <- compute_modal_source(enc_year1)   # year-1 window
pat_source_wholerecord <- compute_modal_source(enc_source_181_assigned)  # whole-record fallback

# Combine: year-1 modal wins; fallback to whole-record when patient has no year-1 encounters
pat_source_181 <- year1_bounds |>
  dplyr::select(ID) |>
  dplyr::left_join(pat_source_year1, by = "ID") |>
  dplyr::left_join(
    pat_source_wholerecord |> dplyr::rename(modal_source_whole = modal_source),
    by = "ID"
  ) |>
  dplyr::mutate(
    SOURCE_analytic_pat = dplyr::coalesce(modal_source, modal_source_whole)
  ) |>
  dplyr::select(ID, SOURCE_analytic_pat)

n_pat_source_missing <- sum(is.na(pat_source_181$SOURCE_analytic_pat))
message(glue("[R/181] Patient SOURCE: {sum(!is.na(pat_source_181$SOURCE_analytic_pat))} assigned, ",
             "{n_pat_source_missing} missing (no encounters in year-1 or whole record)"))
```
