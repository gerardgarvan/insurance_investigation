# Modal SOURCE_analytic in year-1 window (D-07: modal, ties by most recent ADMIT_DATE, then alpha)
compute_modal_source <- function(enc_win, group_col = "ID",
                                  source_col = "SOURCE_analytic", date_col = "ADMIT_DATE") {
  enc_win |>
    dplyr::group_by(dplyr::across(dplyr::all_of(group_col)),
                    dplyr::across(dplyr::all_of(source_col))) |>
    dplyr::summarise(
      n         = dplyr::n(),
      last_date = max(.data[[date_col]]),
      .groups   = "drop"
    ) |>
    dplyr::group_by(dplyr::across(dplyr::all_of(group_col))) |>
    dplyr::slice_max(order_by = n, n = 1L, with_ties = TRUE) |>
    dplyr::arrange(dplyr::across(dplyr::all_of(group_col)),
                   dplyr::desc(last_date),
                   dplyr::across(dplyr::all_of(source_col))) |>
    dplyr::slice_head(n = 1L) |>
    dplyr::ungroup() |>
    dplyr::rename(modal_source = dplyr::all_of(source_col)) |>
    dplyr::select(dplyr::all_of(group_col), modal_source)
}

pat_source_year1       <- compute_modal_source(enc_year1)
pat_source_wholerecord <- compute_modal_source(enc_source_181_assigned)

pat_source_181 <- year1_bounds |>
  dplyr::select(ID) |>

```r
# ==============================================================================
# REGRESSION CHECK (D-11): Pre-existing sheets must match prior R/181 workbook
# ==============================================================================
if (!file.exists(PRIOR_181_WB_PATH)) {
  stop(
    "[R/181] Prior workbook for regression check not found: ", PRIOR_181_WB_PATH,
    "\nSet env var R1923_PRIOR_181_WB to the correct path."
  )
}

prior_181_sheets <- openxlsx::getSheetNames(PRIOR_181_WB_PATH)
message("[R/181] Regression check: prior workbook sheets: ", paste(prior_181_sheets, collapse = ", "))

regression_181_pass  <- TRUE
regression_181_notes <- character(0)

# --- Check B_rates sheet: N, N_CBC, CBC_rate ---
# B_rates columns: Analysis, Group, N, N_CBC, CBC_rate  (all character: sup()/show_rate())
# Two rows per analysis definition (Not far / Far); primary definition is year_index >= 2.
if ("B_rates" %in% prior_181_sheets) {
  prior_brates   <- openxlsx::read.xlsx(PRIOR_181_WB_PATH, sheet = "B_rates")
  current_brates <- rates_df  # produced by the existing B_rates section before workbook creation

  char_check_cols <- c("Analysis", "Group", "N", "N_CBC", "CBC_rate")
  for (col in char_check_cols) {
    if (!col %in% names(prior_brates) || !col %in% names(current_brates)) next
    pv  <- as.character(prior_brates[[col]])
    cv  <- as.character(current_brates[[col]])
    bad <- which(!is.na(pv) & !is.na(cv) & pv != cv)
    if (length(bad) > 0L) {
      regression_181_pass <- FALSE
      for (i in bad) {
        regression_181_notes <- c(regression_181_notes,
          glue::glue("FAIL: B_rates row {i} col {col}: prior='{pv[i]}' new='{cv[i]}'"))
      }
    }
  }
} else {
  regression_181_notes <- c(regression_181_notes, "WARN: B_rates sheet not found in prior workbook")
}

# --- Check A_2x2 sheet: OR, CI bounds from primary analysis ---
# The A_2x2 sheet is a formatted block (not a standard data frame).
# Instead, compare the primary analysis OR and CI from the in-memory results objects.
# The prior workbook's B_rates is the primary numeric check; also compare
# res$primary OR and CI from the analysis definitions:
#   res_primary$or, res_primary$ci_lo, res_primary$ci_hi, res_primary$p
# (variable names determined by reading the analysis loop in R/181 — confirm exact names)
#
# If the prior workbook has an additional sheet with OR/CI (check the QC sheet):
if ("QC" %in% prior_181_sheets) {
  prior_qc   <- openxlsx::read.xlsx(PRIOR_181_WB_PATH, sheet = "QC")
  current_qc <- qc_df  # produced by the existing QC section

  # Compare Check and Value columns
  if (all(c("Check", "Value") %in% names(prior_qc)) &&
      all(c("Check", "Value") %in% names(current_qc))) {
    # Match rows by Check label
    for (i in seq_len(nrow(prior_qc))) {
      chk_label <- as.character(prior_qc$Check[i])
      match_row  <- which(as.character(current_qc$Check) == chk_label)
      if (length(match_row) == 0L) next
      pv <- as.character(prior_qc$Value[i])
      cv <- as.character(current_qc$Value[match_row[1L]])
      if (!is.na(pv) && !is.na(cv) && pv != cv) {
        # For numeric QC values (Ns), try numeric comparison
        pn <- suppressWarnings(as.numeric(pv))
        cn <- suppressWarnings(as.numeric(cv))
        if (!is.na(pn) && !is.na(cn)) {
          if (abs(pn - cn) >= 1e-9) {
            regression_181_pass <- FALSE
            regression_181_notes <- c(regression_181_notes,
              glue::glue("FAIL: QC '{chk_label}': prior={pv} new={cv} diff={abs(pn-cn)}"))
          }
        } else if (pv != cv) {
          regression_181_pass <- FALSE
          regression_181_notes <- c(regression_181_notes,
            glue::glue("FAIL: QC '{chk_label}': prior='{pv}' new='{cv}'"))
        }
      }
    }
  }
} else {
  regression_181_notes <- c(regression_181_notes, "WARN: QC sheet not found in prior workbook")
}

if (!regression_181_pass) {
  stop(
    "[R/181] REGRESSION CHECK FAILED — pre-existing sheets differ from prior workbook.\n",
    paste(regression_181_notes, collapse = "\n"),
    "\nManual sign-off required."
  )
}
message("[R/181] Regression check PASSED.")
```

The comparison uses exact string equality on character fields (all R/181 B_rates and QC
values are stored as character by sup()/show_rate()). For any numeric QC values (Ns) that
happen to be stored as numbers, the check coerces to numeric and applies abs diff < 1e-9.
