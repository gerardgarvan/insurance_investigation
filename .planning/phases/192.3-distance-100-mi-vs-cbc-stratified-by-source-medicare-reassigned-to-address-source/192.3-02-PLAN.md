# ------------------------------------------------------------------------------
# assign_analytic_source() -- Reassign Medicare-sourced encounter SOURCE for
# stratification analysis (Phase 192.3)
#
# Applies a three-step reassignment chain to encounters whose SOURCE_raw is in
# config$claims_source_codes:
#   1. Address lookup: LDS_ADDRESS_HISTORY row in effect at ADMIT_DATE (backward
#      rule: exact interval match, then most recent before ADMIT_DATE). Tie on
#      ADDRESS_PERIOD_START resolved by latest ADDRESS_PERIOD_END, then
#      lexical order on ADDRESS_PERIOD_END. Fall through when: no match, or
#      address SOURCE is missing/blank, or address SOURCE is itself in
#      config$claims_source_codes.
#   2. Modal non-claims encounter SOURCE across the WHOLE record (all ENCOUNTER
#      rows whose SOURCE_raw is NOT in config$claims_source_codes). Ties broken
#      by most recent ADMIT_DATE, then alphabetical SOURCE. Fall through when no
#      qualifying encounters exist.
#   3. Unresolved: SOURCE_analytic = "Medicare_unresolved". Kept; not dropped.
#
# @param enc         Tibble with columns: ID (chr), ENCOUNTERID (chr),
#                    ADMIT_DATE (Date), SOURCE (chr). Must contain ALL encounters

#                    contain: ID, ADDRESS_PERIOD_START, ADDRESS_PERIOD_END, SOURCE.
#                    Passed as argument for testability (same seam as get_zip9_at_date).

**Step 1: Address lookup for claims rows**

Parse date columns in addr_full:
```r
  addr <- addr_full |>
    dplyr::mutate(
      period_start = as.Date(ADDRESS_PERIOD_START),
      period_end   = dplyr::if_else(
        is.na(ADDRESS_PERIOD_END) | trimws(as.character(ADDRESS_PERIOD_END)) == "",
        as.Date("9999-12-31"),
        as.Date(ADDRESS_PERIOD_END)
      ),
      addr_source  = trimws(SOURCE)
    )
```

For each claims encounter (ID + ADMIT_DATE), find the in-effect address row:
- Match: `period_start <= ADMIT_DATE` AND `ADMIT_DATE < period_end`
- If multiple matches: break ties by latest period_start, then latest period_end (as.character for lex), then row order
- Accept match only when `addr_source` is non-blank AND NOT in config$claims_source_codes

Implement via a left_join + filter + group_by approach:
```r
  claims_addr <- enc_claims |>
    dplyr::select(ID, ENCOUNTERID, ADMIT_DATE) |>
    dplyr::left_join(
      addr |> dplyr::select(ID, period_start, period_end, addr_source),
      by = "ID",
      relationship = "many-to-many"
    ) |>
    dplyr::filter(
      !is.na(period_start),
      period_start <= ADMIT_DATE,
      ADMIT_DATE < period_end
    ) |>
    # Accept only non-blank, non-claims address SOURCE
    dplyr::filter(
      !is.na(addr_source), addr_source != "",
      !addr_source %in% config$claims_source_codes
    ) |>
    # Tiebreak: latest period_start, then latest period_end (lex), then row order
    dplyr::arrange(ID, ENCOUNTERID, dplyr::desc(period_start), dplyr::desc(period_end)) |>
    dplyr::group_by(ENCOUNTERID) |>
    dplyr::slice_head(n = 1L) |>
    dplyr::ungroup() |>
    dplyr::select(ENCOUNTERID, source_from_address = addr_source)
```

If no interval match: also run a "most recent before" pass (period_start <= ADMIT_DATE,
any period_end) for the fallback — combine interval + most-recent-before as one lookup,
preferring interval over most-recent-before:

```r
  # Most-recent-before fallback (no period_end constraint):
  claims_addr_fallback <- enc_claims |>
    dplyr::select(ID, ENCOUNTERID, ADMIT_DATE) |>
    dplyr::left_join(
      addr |> dplyr::select(ID, period_start, period_end, addr_source),
      by = "ID",
      relationship = "many-to-many"
    ) |>
    dplyr::filter(
      !is.na(period_start),
      period_start <= ADMIT_DATE,
      !is.na(addr_source), addr_source != "",
      !addr_source %in% config$claims_source_codes
    ) |>
    dplyr::arrange(ID, ENCOUNTERID, dplyr::desc(period_start), dplyr::desc(period_end)) |>
    dplyr::group_by(ENCOUNTERID) |>
    dplyr::slice_head(n = 1L) |>
    dplyr::ungroup() |>
    dplyr::select(ENCOUNTERID, source_from_address_fb = addr_source)

  # Coalesce: interval match wins; fallback when no interval match
  enc_claims_step1 <- enc_claims |>
    dplyr::left_join(claims_addr,          by = "ENCOUNTERID") |>
    dplyr::left_join(claims_addr_fallback, by = "ENCOUNTERID") |>
    dplyr::mutate(
      addr_resolved = dplyr::coalesce(source_from_address, source_from_address_fb)
    )
```

**Step 2: Modal non-claims encounter SOURCE (whole record)**

Build whole-record non-claims SOURCE for every patient:
```r
  whole_record_nonclaims <- enc |>
    dplyr::filter(!SOURCE_raw %in% config$claims_source_codes,
                  !is.na(SOURCE_raw), SOURCE_raw != "") |>
    dplyr::select(ID, SOURCE_raw, ADMIT_DATE)

  # Modal with tiebreak: most recent ADMIT_DATE, then alphabetical SOURCE
  modal_tbl <- whole_record_nonclaims |>
    dplyr::group_by(ID, SOURCE_raw) |>
    dplyr::summarise(n = dplyr::n(), last_date = max(ADMIT_DATE), .groups = "drop") |>
    dplyr::group_by(ID) |>
    dplyr::slice_max(order_by = n, n = 1L, with_ties = TRUE) |>
    # Among ties on frequency: pick most recent last_date, then alphabetical SOURCE
    dplyr::arrange(ID, dplyr::desc(last_date), SOURCE_raw) |>
    dplyr::slice_head(n = 1L) |>
    dplyr::ungroup() |>
    dplyr::select(ID, modal_source = SOURCE_raw)
```

**Resolve all three steps:**
```r
  enc_claims_resolved <- enc_claims_step1 |>
    dplyr::left_join(modal_tbl, by = "ID") |>
    dplyr::mutate(
      SOURCE_analytic = dplyr::case_when(
        !is.na(addr_resolved)   ~ addr_resolved,
        !is.na(modal_source)    ~ modal_source,
        TRUE                    ~ "Medicare_unresolved"
      ),
      source_reassign_method = dplyr::case_when(
        !is.na(addr_resolved)   ~ "address",
        !is.na(modal_source)    ~ "modal_encounter",
        TRUE                    ~ "unresolved"
      ),
      source_reassigned = 1L
    ) |>
    dplyr::select(-addr_resolved, -source_from_address, -source_from_address_fb, -modal_source)
```

**Recombine and return:**
```r
  dplyr::bind_rows(enc_nonclaims, enc_claims_resolved) |>
    dplyr::arrange(ENCOUNTERID)
}
```
