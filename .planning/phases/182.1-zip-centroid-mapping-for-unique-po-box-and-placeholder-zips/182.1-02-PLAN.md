```r
# ---------------------------------------------------------------------------
# Phase 182.1: Crosswalk fallback tiers for zip_not_in_db encounter pairs
# Tier 1 (zipcodeR_raw): already computed above -- existing zip_distance() call
# Tier 2 (zipcodeR_crosswalk): crosswalk facility ZIP to ZCTA, retry zip_distance()
# Tier 3 (haversine_gazetteer): haversine from gazetteer centroids
# Only applies to rows where Tier 1 returned NA distance.
# ---------------------------------------------------------------------------

# Initialize distance_method for all rows
enc_dist <- enc_dist |>
  dplyr::mutate(
    distance_method = dplyr::case_when(
      !is.na(distance_km) ~ "zipcodeR_raw",
      TRUE                ~ NA_character_
    )
  )

if (HRSA_XWALK_AVAILABLE) {
  # --- Tier 2: zipcodeR_crosswalk ---
  na_rows <- is.na(enc_dist$distance_km)
  message(glue("  [crosswalk tier 2] {sum(na_rows)} NA-distance rows eligible for crosswalk retry"))

  if (sum(na_rows) > 0L) {
    # Crosswalk facility ZIP to ZCTA
    zcta_facility <- map_zip_to_zcta(enc_dist$zip5_facility[na_rows], hrsa_xwalk)
    # Also crosswalk patient ZIP to ZCTA (may help if patient ZIP is also unique)
    zcta_patient  <- map_zip_to_zcta(enc_dist$zip5_patient[na_rows],  hrsa_xwalk)

    # Retry zip_distance() using ZCTAs where available
    dist_t2 <- mapply(function(zf, zp) {
      zf_use <- if (!is.na(zf)) zf else enc_dist$zip5_facility[na_rows][match(zf, zcta_facility)]
      zp_use <- if (!is.na(zp)) zp else enc_dist$zip5_patient[na_rows][match(zp, zcta_patient)]
      tryCatch(
        zipcodeR::zip_distance(
          if (!is.na(zcta_facility[match(zf, zcta_facility)])) zcta_facility[match(zf, zcta_facility)] else enc_dist$zip5_facility[na_rows][match(zf, zcta_facility)],
          if (!is.na(zcta_patient[match(zp, zcta_patient)]))  zcta_patient[match(zp, zcta_patient)]  else enc_dist$zip5_patient[na_rows][match(zp, zcta_patient)],
          lonlat = FALSE
        )$distance,
        error = function(e) NA_real_
      )
    }, zcta_facility, zcta_patient, SIMPLIFY = TRUE)
```
