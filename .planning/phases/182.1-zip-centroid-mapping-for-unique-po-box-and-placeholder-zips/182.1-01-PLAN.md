# ==============================================================================
# utils_zip_crosswalk.R -- HRSA ZIP-to-ZCTA crosswalk helpers (Phase 182.1)
# ==============================================================================
#
# Purpose:
#   Crosswalk facility ZIP5 codes to ZCTA using the HRSA UDS Mapper crosswalk,
#   and detect placeholder facility ZIP5 codes (000xx patterns and ZIP3+"00"
#   codes absent from the crosswalk).
#
# Inputs (to map_zip_to_zcta):
#   zip5:  character vector of 5-digit ZIP codes
#   xwalk: data frame with at least columns ZIP and ZCTA (character)
#          loaded from CONFIG$hrsa_zip_zcta_path
#
# Inputs (to is_placeholder_facility_zip5):
#   zip5:  character vector of 5-digit ZIP codes
#   xwalk: same crosswalk data frame as above
#
# Outputs:
#   map_zip_to_zcta()             -> character vector (ZCTA or NA)
#   is_placeholder_facility_zip5() -> logical vector
#
# Design decisions (Phase 182.1 CONTEXT.md):
#   D-04: Two pure functions; no hard-coded ZIP lists.
#   D-05: is_sentinel_zip5() (patient-side) is NOT used or modified here.
#   D-06: Placeholder step runs before FACILITYID fill in R/122 (orchestrated
#         there, not here).
#
# Dependencies:
#   dplyr, stringr (no library() calls -- caller sources R/00_config.R first)
#   normalize_zip5_raw() from R/utils/utils_address.R (auto-loaded)
#
# No I/O, no library() calls, no data.table syntax.
# ==============================================================================

#' Map a ZIP5 to its ZCTA using the HRSA UDS Mapper crosswalk.
#'
#' @param zip5  Character vector of 5-digit ZIP codes (raw; normalized internally).
#' @param xwalk Data frame with character columns ZIP and ZCTA loaded from
#'              CONFIG$hrsa_zip_zcta_path.
#' @return Character vector (same length as zip5). ZCTA string where found,
#'         NA_character_ where not found or where input normalizes to NA.
map_zip_to_zcta <- function(zip5, xwalk) {
  stopifnot(
    is.data.frame(xwalk),
    all(c("ZIP", "ZCTA") %in% names(xwalk))
  )
  z5 <- normalize_zip5_raw(zip5)
  # Left-join to crosswalk; return ZCTA column
  idx <- match(z5, xwalk$ZIP)
  result <- xwalk$ZCTA[idx]
  # Coerce empty-string ZCTA to NA (some crosswalk files use "" for no match)
  dplyr::if_else(is.na(result) | result == "", NA_character_, result)
}

#' Detect placeholder facility ZIP5 codes.
#'
#' A facility ZIP5 is a placeholder if:
#'   (1) It matches the 000xx pattern (first 3 digits are "000"), OR
#'   (2) It ends in "00" (ZIP3+"00" pattern, e.g. "32800", "32700", "34700")
#'       AND is absent from the HRSA crosswalk (not a real delivery ZIP).
#'
#' The crosswalk-presence check prevents false positives: a real ZIP that
#' happens to end in "00" will be in the crosswalk and will NOT be flagged.
#'
#' @param zip5  Character vector of 5-digit ZIP codes (raw; normalized internally).
#' @param xwalk Data frame with character column ZIP (from CONFIG$hrsa_zip_zcta_path).
#' @return Logical vector (same length as zip5). TRUE = placeholder, FALSE = real,
#'         NA = input was NA or failed normalization.
is_placeholder_facility_zip5 <- function(zip5, xwalk) {
  stopifnot(
    is.data.frame(xwalk),
    "ZIP" %in% names(xwalk)
  )
  z5 <- normalize_zip5_raw(zip5)
  in_xwalk <- z5 %in% xwalk$ZIP

  pattern_000xx     <- !is.na(z5) & stringr::str_detect(z5, "^000")
  pattern_zip3_00   <- !is.na(z5) & stringr::str_detect(z5, "00$") & !in_xwalk

  result <- dplyr::if_else(
    is.na(z5),
    NA,
    pattern_000xx | pattern_zip3_00
  )
  result
}
