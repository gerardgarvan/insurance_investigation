    haversine_km <- function(lat1, lon1, lat2, lon2) {
      R   <- 6371
      d2r <- pi / 180
      lat1r <- lat1 * d2r; lat2r <- lat2 * d2r
      dlat  <- (lat2 - lat1) * d2r
      dlon  <- (lon2 - lon1) * d2r

      2 * R * asin(pmin(1, sqrt(a)))
    }
