```r
# Prefix match of any key length; DX_TYPE gated by system.
is_autoimmune_code <- function(dx, dx_type,
                               codes  = AUTOIMMUNE_DX_CODES,
                               accept = AUTOIMMUNE_DX_TYPE_ACCEPT) {
  n <- length(dx); if (n == 0) return(logical(0))
  dx_clean <- normalize_icd(trimws(dx))
  dt <- trimws(as.character(dx_type))
  match_any <- function(x, keys) {
    if (!length(keys)) return(rep(FALSE, length(x)))
    Reduce(`|`, lapply(keys, function(k) startsWith(x, k)))
  }
  out <- rep(FALSE, n)
  i10 <- !is.na(dx_clean) & !is.na(dt) & dt %in% accept$icd10
  i09 <- !is.na(dx_clean) & !is.na(dt) & dt %in% accept$icd9
  out[i10] <- match_any(dx_clean[i10], names(codes$icd10))
  out[i09] <- match_any(dx_clean[i09], names(codes$icd9))
  out
}

# Pure: in-memory diagnosis rows -> one row per cohort ID.
# dx: data.frame(ID, DX, DX_TYPE, DX_DATE, ADMIT_DATE); cohort_ids: character
summarise_autoimmune_dx <- function(dx, cohort_ids, sentinel) {
  q <- dx[is_autoimmune_code(dx$DX, dx$DX_TYPE), , drop = FALSE]
  d1 <- as.Date(q$DX_DATE); d2 <- as.Date(q$ADMIT_DATE)
  ok1 <- !is.na(d1) & d1 > sentinel; ok2 <- !is.na(d2) & d2 > sentinel
  q$valid_date <- as.Date(ifelse(ok1, d1, ifelse(ok2, d2, NA)), origin = "1970-01-01")
  per_id <- q |>
    dplyr::group_by(ID) |>
    dplyr::summarise(
      first_autoimmune_dx_date = if (all(is.na(valid_date))) as.Date(NA) else min(valid_date, na.rm = TRUE),
      any_qualifying  = TRUE,
      any_valid_date  = any(!is.na(valid_date)),
      .groups = "drop")
  out <- tibble::tibble(ID = unique(cohort_ids)) |>
    dplyr::left_join(per_id, by = "ID") |>
    dplyr::mutate(
      has_autoimmune_dx = !is.na(first_autoimmune_dx_date),
      qc_qualifying_no_valid_date = dplyr::coalesce(any_qualifying, FALSE) & !dplyr::coalesce(any_valid_date, FALSE))
  stopifnot(nrow(out) == dplyr::n_distinct(out$ID),
            !any(out$has_autoimmune_dx & is.na(out$first_autoimmune_dx_date)),
            !any(!out$has_autoimmune_dx & !is.na(out$first_autoimmune_dx_date)))
  out
}

# DuckDB wrapper: cohort registered and joined in SQL (R/169 pattern), no IN list, no full collect.
build_autoimmune_dx_flags <- function(cohort_ids, sentinel) {
  opened_here <- FALSE
  if (!exists("pcornet_con", envir = .GlobalEnv) || !DBI::dbIsValid(pcornet_con)) {
    open_pcornet_con(); opened_here <- TRUE }
  duckdb::duckdb_register(pcornet_con, "aim_cohort", tibble::tibble(ID = unique(cohort_ids)))
  on.exit({ duckdb::duckdb_unregister(pcornet_con, "aim_cohort")
            if (opened_here) close_pcornet_con() }, add = TRUE)
  dx <- DBI::dbGetQuery(pcornet_con, "
    SELECT d.ID, d.DX, d.DX_TYPE, d.DX_DATE, d.ADMIT_DATE
    FROM DIAGNOSIS d JOIN aim_cohort c ON c.ID = d.ID")
  summarise_autoimmune_dx(dx, cohort_ids, sentinel)
}
```
