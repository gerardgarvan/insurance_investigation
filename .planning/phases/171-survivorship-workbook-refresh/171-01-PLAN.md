**G. Sheet-comparison helper**
Read sheet names at runtime from the reference workbook (do not hardcode column names).
```r
compare_sheet <- function(sheet_name, label) {
  tryCatch({
    ref_df <- read_excel(ref_path, sheet = sheet_name)
    new_df <- read_excel(new_path, sheet = sheet_name)
    if (identical(ref_df, new_df)) {
      cat(sprintf("[PASS] %s: identical to reference\n", label))
      n_pass <<- n_pass + 1L
    } else {
      diffs <- sum(!mapply(identical, ref_df, new_df))
      cat(sprintf("[FAIL] %s: %d column(s) differ\n", label, diffs))
      n_fail <<- n_fail + 1L
    }
  }, error = function(e) {
    cat(sprintf("[FAIL] %s: error reading sheet: %s\n", label, conditionMessage(e)))
    n_fail <<- n_fail + 1L
  })
}
```

if (ref_ok && new_ok) {
  # Check 1: A_rates_summary — 14 modalities, primary-tier
  compare_sheet("A_rates_summary",   "A_rates_summary (14 modalities)")

  # Check 2: B_rates_by_fu_year — 14 modalities, primary-tier by FU year
  compare_sheet("B_rates_by_fu_year", "B_rates_by_fu_year (14 modalities)")

  # Check 3: C_anthracycline_echo — primary counts; echo is a changed modality
  # but change was any-tier/sensitivity only; primary counts must be identical
  compare_sheet("C_anthracycline_echo", "C_anthracycline_echo (primary counts)")

  # Check 4: QC interval reconciliation
  # Sum of interval person-years must equal total person-years (tolerance < 0.01 days)
  tryCatch({
    qc_ref <- read_excel(ref_path, sheet = "QC")
    qc_new <- read_excel(new_path, sheet = "QC")
    # Look for a column containing "person_years" or "py" (runtime discovery)
    py_col <- grep("person.year|^py$", names(qc_new), ignore.case = TRUE, value = TRUE)[1]
    if (is.na(py_col)) {
      cat("[FAIL] QC_interval_reconciliation: no person-year column found in QC sheet\n")
      n_fail <- n_fail + 1L
    } else {
      total_py  <- qc_new[[py_col]][grep("total", qc_new[[1]], ignore.case = TRUE)[1]]
      sum_py    <- sum(qc_new[[py_col]][grep("total", qc_new[[1]], ignore.case = TRUE, invert = TRUE)],
                       na.rm = TRUE)
      delta_days <- abs(total_py - sum_py) * 365.25
      if (delta_days < 0.01) {
        cat(sprintf("[PASS] QC_interval_reconciliation: delta %.4f days (< 0.01)\n", delta_days))
        n_pass <- n_pass + 1L
      } else {
        cat(sprintf("[FAIL] QC_interval_reconciliation: delta %.4f days (>= 0.01)\n", delta_days))
        n_fail <- n_fail + 1L
      }
    }
  }, error = function(e) {
    cat(sprintf("[FAIL] QC_interval_reconciliation: %s\n", conditionMessage(e)))
    n_fail <<- n_fail + 1L
  })
} else {
  cat("[SKIP] sheet checks skipped — one or both workbooks missing\n")
}
