    read_txt <- function(path, sheet) read_excel(path, sheet = sheet, col_types = "text", .name_repair = "minimal")

      if (!both) return("FAIL — input workbook missing")
      r <- read_txt(ref_path, sheet); n <- read_txt(new_path, sheet)
      if (!identical(dim(r), dim(n))) return(sprintf("FAIL — dims ref %dx%d vs new %dx%d", nrow(r), ncol(r), nrow(n), ncol(n)))
      if (!identical(names(r), names(n))) return("FAIL — column names differ")
      rm <- as.matrix(r); nm <- as.matrix(n)
      same <- (is.na(rm) & is.na(nm)) | (!is.na(rm) & !is.na(nm) & rm == nm)
      diff_cells <- sum(!same)
      if (diff_cells == 0) "PASS — identical cell text" else sprintf("FAIL — %d cell(s) differ", diff_cells)
