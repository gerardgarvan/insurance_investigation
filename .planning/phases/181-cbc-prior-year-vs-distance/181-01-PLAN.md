    run_2x2 <- function(df, label) {
      m <- matrix(as.numeric(table(factor(df$far, 0:1), factor(df$cbc_year, 0:1))), 2)  # rows not far/far; cols no/yes
      ex <- suppressWarnings(chisq.test(m, correct = FALSE)$expected)
      use_fisher <- any(ex < 5)
      p <- if (use_fisher) fisher.test(m)$p.value else chisq.test(m, correct = FALSE)$p.value
      cells <- c(m[1,1], m[1,2], m[2,1], m[2,2])
      adj <- if (any(cells == 0)) 0.5 else 0
      or <- crude_or(m[1,1], m[1,2], m[2,1], m[2,2])
      se <- sqrt(sum(1 / (cells + adj)))
      list(label = label, n = sum(m), m = m, min_expected = min(ex),
           test = if (use_fisher) "Fisher's exact" else "Pearson chi-squared",
           reason = glue::glue("min expected cell = {round(min(ex), 1)}"),
           p = p, or = or, ci_lo = exp(log(or) - 1.96 * se), ci_hi = exp(log(or) + 1.96 * se),
           small_cell = any(cells >= 1 & cells <= 10))
    }
