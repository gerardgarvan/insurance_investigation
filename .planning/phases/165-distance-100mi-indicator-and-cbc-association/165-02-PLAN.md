    **Section 3: Dispatch — primary bivariate analysis**

    Implement a dispatch block using METHOD. For each branch, compute: OR, lower_ci, upper_ci, se_log_or, p_value, and (if applicable) deff. Store in a named list `primary_result`.

    ```r
    primary_result <- if (METHOD == "gee") {
      enc_gee <- enc_analysis %>% dplyr::arrange(ID)
      fit <- geepack::geeglm(
        cbc_on_date ~ far_from_care_100mi,
        data = enc_gee, id = ID,
        family = binomial(link = "logit"), corstr = "exchangeable"
      )
      s    <- summary(fit)
      beta <- coef(fit)["far_from_care_100mi"]
      se   <- s$coefficients["far_from_care_100mi", "Std.err"]
      p    <- s$coefficients["far_from_care_100mi", "Pr(>|W|)"]
      list(method = "GEE (exchangeable)", or = exp(beta),
           lower_ci = exp(beta - 1.96*se), upper_ci = exp(beta + 1.96*se),
           se_log_or = se, p_value = p, deff = NA_real_,
           n_enc = nrow(enc_gee), n_pat = dplyr::n_distinct(enc_gee$ID),
           gee_alpha = fit$geese$alpha)

    } else if (METHOD == "rao_scott") {
      svy <- survey::svydesign(ids = ~ID, data = enc_analysis, weights = ~1)
      svy_fit <- survey::svyglm(cbc_on_date ~ far_from_care_100mi,
                                design = svy, family = quasibinomial(link = "logit"))
      beta <- coef(svy_fit)["far_from_care_100mi"]
      se   <- summary(svy_fit)$coefficients["far_from_care_100mi", "Std. Error"]
      ci   <- confint(svy_fit)["far_from_care_100mi", ]
      p    <- summary(svy_fit)$coefficients["far_from_care_100mi", "Pr(>|t|)"]
      # DEFF vs naive
      ct   <- table(enc_analysis$far_from_care_100mi, enc_analysis$cbc_on_date)
      a <- ct[2,2]; b <- ct[2,1]; c_c <- ct[1,2]; d <- ct[1,1]
      se_naive <- sqrt(1/a + 1/b + 1/c_c + 1/d)
      deff     <- (se / se_naive)^2
      list(method = "Rao-Scott (survey::svyglm)", or = exp(beta),
           lower_ci = exp(ci[1]), upper_ci = exp(ci[2]),
           se_log_or = se, p_value = p, deff = deff,
           n_enc = nrow(enc_analysis), n_pat = dplyr::n_distinct(enc_analysis$ID),
           gee_alpha = NA_real_)

    } else {  # patient_fisher
      ct_pat <- table(pat_analysis$any_far, pat_analysis$any_cbc_post_anchor)
      ft     <- fisher.test(ct_pat)
      list(method = "Patient-level Fisher exact", or = ft$estimate,
           lower_ci = ft$conf.int[1], upper_ci = ft$conf.int[2],
           se_log_or = NA_real_, p_value = ft$p.value, deff = NA_real_,
           n_enc = NA_integer_, n_pat = nrow(pat_analysis),
           gee_alpha = NA_real_)
    }
    ```

    Run the adjusted model only if METHOD is "gee" or "rao_scott" (encounter-level). Patient-level Fisher does not support covariate adjustment in the same framework; note this in the C_sensitivity sheet.

    Drop patients with missing rurality; record N dropped. SOURCE = enc_dist's site/source column (confirm column name from enc_distance schema; use SOURCE from enc_dist if present, else omit).

    ```r
    # Drop missing rurality for adjusted model
    enc_adj <- enc_analysis %>%
      dplyr::left_join(rurality, by = "ID") %>%
      dplyr::filter(!is.na(ruca_category))
    n_dropped_rurality <- n_pat_analysis - dplyr::n_distinct(enc_adj$ID)

    sensitivity_result <- tryCatch({
      if (METHOD == "gee") {
        enc_gee_adj <- enc_adj %>% dplyr::arrange(ID)
        fit_adj <- geepack::geeglm(
          cbc_on_date ~ far_from_care_100mi + ruca_category + payer_category,
          data = enc_gee_adj, id = ID,
          family = binomial(link = "logit"), corstr = "exchangeable"
        )
        list(fit = fit_adj, method = "GEE adjusted", n_dropped = n_dropped_rurality)
      } else if (METHOD == "rao_scott") {
        svy_adj <- survey::svydesign(ids = ~ID, data = enc_adj, weights = ~1)
        fit_adj <- survey::svyglm(
          cbc_on_date ~ far_from_care_100mi + ruca_category + payer_category,
          design = svy_adj, family = quasibinomial(link = "logit")
        )
        list(fit = fit_adj, method = "Rao-Scott adjusted", n_dropped = n_dropped_rurality)
      } else {
        list(fit = NULL, method = "Patient-level Fisher — covariate adjustment not applicable",
             n_dropped = n_dropped_rurality)
      }
    }, error = function(e) {
      list(fit = NULL, method = paste("Adjusted model failed:", conditionMessage(e)),
           n_dropped = n_dropped_rurality)
    })
    ```

    ```r
    openxlsx::addWorksheet(wb, "C_sensitivity")
    if (!is.null(sensitivity_result$fit)) {
      s_coef <- summary(sensitivity_result$fit)$coefficients
      c_df <- as.data.frame(s_coef) %>%
        tibble::rownames_to_column("Covariate") %>%
        dplyr::rename(Estimate = 2, SE = 3) %>%
        dplyr::mutate(
          OR        = round(exp(Estimate), 3),
          CI_lower  = round(exp(Estimate - 1.96*SE), 3),
          CI_upper  = round(exp(Estimate + 1.96*SE), 3)
        )
      openxlsx::writeData(wb, "C_sensitivity",
                          tibble::tibble(note = glue(
                            "{sensitivity_result$method}. N patients dropped for missing rurality: {n_dropped_rurality}."
                          )))
      openxlsx::writeData(wb, "C_sensitivity", c_df, startRow = 3)
    } else {
      openxlsx::writeData(wb, "C_sensitivity",
                          tibble::tibble(note = sensitivity_result$method))
    }

    # ---- QC sheet ----
    openxlsx::addWorksheet(wb, "QC")
    qc_df <- tibble::tribble(
      ~Item,          ~N,
      "enc_distance input rows (computed status)", suppress_small(n_enc_input),
      "Analysis set encounters (non-NA far_from_care_100mi)", suppress_small(n_enc_analysis),
      "Encounters with same-day CBC", suppress_small(n_enc_with_cbc),
      "Patients in analysis set", suppress_small(n_pat_analysis),
      "Patients with any post-anchor CBC", suppress_small(n_pat_with_cbc),
      "Telehealth/virtual encounters (ENC_TYPE TH/OA)", suppress_small(n_telehealth),
      "Patients missing rurality (dropped from sensitivity model)", suppress_small(n_missing_rur)
    )
    openxlsx::writeData(wb, "QC", qc_df)
    ```
