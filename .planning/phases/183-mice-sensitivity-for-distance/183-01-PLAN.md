**Step 3a — Fit complete-case GEE on pinned RDS**

Use the complete-case subset of py_full (same as Phase 179 primary analysis):
```r
py_cc <- py_full |>
  filter(year_index >= 2L, !exposure_missing, !is.na(far)) |>
  arrange(ID, year_index) |>
  mutate(
    year_index_cat = factor(if_else(year_index >= 5L, "5+", as.character(year_index)),
                            levels = c("1", "2", "3", "4", "5+")),
    year_index_cat = droplevels(year_index_cat),
    cl = as.integer(factor(ID, levels = unique(ID)))
  )
stopifnot("Task-0 complete-case set is empty" = nrow(py_cc) > 0L)
message(glue("  Task-0 complete-case: {nrow(py_cc)} py, {n_distinct(py_cc$ID)} patients"))

t0_fit <- geepack::geeglm(
  cbc_year ~ far + year_index_cat,
  data   = py_cc,
  id     = cl,
  family = binomial("logit"),
  corstr = GEE_CORSTR
)
t0_cf  <- summary(t0_fit)$coefficients
t0_b   <- t0_cf["far", "Estimate"]
t0_se  <- t0_cf["far", "Std.err"]   # robust SE
t0_or  <- exp(t0_b)
t0_lo  <- exp(t0_b - 1.96 * t0_se)
t0_hi  <- exp(t0_b + 1.96 * t0_se)
message(glue("  Task-0 GEE: OR {round(t0_or, 3)} ({round(t0_lo, 3)}-{round(t0_hi, 3)})"))
```
