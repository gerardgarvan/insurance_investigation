    # Rubin (1987); df per Barnard & Rubin form for large complete-data df
    pool_rubin <- function(Q, SE) {
      m <- length(Q); U <- mean(SE^2); B <- var(Q); Tv <- U + (1 + 1/m) * B
      r <- (1 + 1/m) * B / U
      df <- if (B > 0) (m - 1) * (1 + 1/r)^2 else Inf
      q <- qt(0.975, df)
      list(or = exp(mean(Q)), lo = exp(mean(Q) - q * sqrt(Tv)), hi = exp(mean(Q) + q * sqrt(Tv)),
           fmi = (r + 2/(df + 3)) / (r + 1), df = df, U = U, B = B, T = Tv, m = m)
    }
