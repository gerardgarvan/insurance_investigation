```r
fit_gee_imputed <- function(py_imp, year_subset_label, year_filter_fn) {
  d <- py_imp |>
    year_filter_fn() |>
    filter(!is.na(far)) |>
    arrange(ID, year_index) |>
    mutate(
      year_index_cat = factor(if_else(year_index >= 5L, "5+", as.character(year_index)),
                              levels = c("1", "2", "3", "4", "5+")),
      year_index_cat = droplevels(year_index_cat),
      cl = as.integer(factor(ID, levels = unique(ID)))
    )
  fit <- tryCatch(
    geepack::geeglm(cbc_year ~ far + year_index_cat, data = d, id = cl,
                    family = binomial("logit"), corstr = GEE_CORSTR),
    error = function(e) e
  )
  if (inherits(fit, "error")) stop(glue("D-09 FAIL: GEE failed in imputed dataset: {conditionMessage(fit)}"))
  if (!is.null(fit$geese$error) && fit$geese$error != 0)
    warning(glue("geese error code {fit$geese$error} in imputed dataset"))
  cf <- summary(fit)$coefficients
  list(Q = cf["far", "Estimate"], V = cf["far", "Std.err"]^2)  # log-OR and robust variance
}
```

Step 5c — Pool via Rubin's rules (cite Rubin 1987 inline):
```r
pool_rubin <- function(Q_vec, V_vec) {
  # Rubin (1987) combining rules for multiply-imputed data
  m     <- length(Q_vec)
  Q_bar <- mean(Q_vec)              # pooled point estimate (log-OR scale)
  U_bar <- mean(V_vec)              # within-imputation variance
  B     <- var(Q_vec)               # between-imputation variance
  T_var <- U_bar + (1 + 1/m) * B   # total variance
  fmi   <- (B + B/m) / T_var       # fraction of missing information
  list(
    log_or   = Q_bar,
    or       = exp(Q_bar),
    ci_lo    = exp(Q_bar - 1.96 * sqrt(T_var)),
    ci_hi    = exp(Q_bar + 1.96 * sqrt(T_var)),
    U_bar    = U_bar, B = B, T_var = T_var, fmi = fmi, m = m
  )
}
```

```r
read_bound_or <- function(xlsx_path, sheet_name, label_hint, or_col_hint = "OR") {
  wb <- readxl::read_excel(xlsx_path, sheet = sheet_name, col_types = "text")
  label_col <- names(wb)[1]
  or_col    <- grep(or_col_hint, names(wb), ignore.case = TRUE, value = TRUE)[1]
  stopifnot("OR column not found" = !is.na(or_col))
  hit <- wb[grepl(label_hint, trimws(wb[[label_col]]), ignore.case = TRUE), ]
  if (nrow(hit) != 1L)
    stop(glue("D-13 FAIL: label matching '{label_hint}' found {nrow(hit)} times ",
              "in sheet '{sheet_name}' of Phase 180 workbook. Expected exactly 1."))
  as.numeric(hit[[or_col]][1])
}
```
