cat("abvd_class changed:", sum(ch, na.rm = TRUE), "\n")
t <- table(old = o$abvd_class[ch], new = n$abvd_class[m][ch], useNA = "ifany"); t[t >= 1 & t <= 10] <- NA; print(t)
