    Define build_facility_zip_lookup(enc_raw) exactly per this skeleton (from 182-RESEARCH.md lines 264-291), using dplyr only (no data.table):

    ```r
    build_facility_zip_lookup <- function(enc_raw) {
      stopifnot(all(c("SOURCE", "FACILITYID", "enc_zip_raw") %in% names(enc_raw)))
      enc_raw %>%
        dplyr::mutate(candidate_zip5 = normalize_zip5_raw(enc_zip_raw)) %>%
        dplyr::group_by(SOURCE, FACILITYID) %>%
        dplyr::summarise(
          n_total         = dplyr::n(),
          n_distinct_zips = dplyr::n_distinct(candidate_zip5[!is.na(candidate_zip5)]),
          n_supporting    = sum(!is.na(candidate_zip5)),
          fill_zip        = dplyr::if_else(
                              n_distinct_zips == 1L & n_supporting >= 3L,
                              candidate_zip5[!is.na(candidate_zip5)][1],
                              NA_character_),
          fill_status     = dplyr::case_when(
                              n_distinct_zips == 1L & n_supporting >= 3L ~ "filled_tier1",
                              n_distinct_zips > 1L                       ~ "ambiguous",
                              TRUE                                       ~ "no_donor"),
          sensitivity_modal_zip = {
            v <- candidate_zip5[!is.na(candidate_zip5)]
            if (length(v) == 0) NA_character_ else names(which.max(table(v)))
          },
          sensitivity_modal_n = {
            v <- candidate_zip5[!is.na(candidate_zip5)]
            if (length(v) == 0) 0L else as.integer(max(table(v)))
          },
          sensitivity_modal_share = {
            v <- candidate_zip5[!is.na(candidate_zip5)]
            if (length(v) == 0) NA_real_ else as.numeric(max(table(v))) / length(v)
          },
          .groups = "drop"
        )
    }
    ```

    IMPORTANT: a group whose candidate_zip5 is all NA has n_distinct_zips == 0 and n_supporting == 0; the case_when falls through to "no_donor" (correct). A group with n_distinct_zips == 1 but n_supporting < 3 ALSO falls through to "no_donor" (neither branch 1 nor branch 2 matches). This is the intended Tier-1 behavior per D-02 (>= 3 supporting required).

    Then define apply_facility_zip_fill(enc_for_dist, lookup):

    ```r
    apply_facility_zip_fill <- function(enc_for_dist, lookup) {
      stopifnot(all(c("SOURCE", "FACILITYID", "zip5_facility") %in% names(enc_for_dist)))
      n0 <- nrow(enc_for_dist)
      out <- enc_for_dist %>%
        dplyr::left_join(
          dplyr::select(lookup, SOURCE, FACILITYID, fill_zip, fill_status),
          by = c("SOURCE", "FACILITYID")
        ) %>%
        dplyr::mutate(
          .was_observed = !is.na(zip5_facility),
          .fillable     = is.na(zip5_facility) & fill_status == "filled_tier1" & !is.na(fill_zip),
          zip5_facility = dplyr::if_else(.fillable, fill_zip, zip5_facility),
          facility_zip_source = dplyr::case_when(
            .was_observed ~ "observed",
            .fillable     ~ "filled_tier1",
            TRUE          ~ "not_filled"
          )
        ) %>%
        dplyr::select(-.was_observed, -.fillable, -fill_zip, -fill_status)
      stopifnot("apply_facility_zip_fill fanned out" = nrow(out) == n0)
      out
    }
    ```
    Keep both functions dependency-light; they rely only on dplyr and normalize_zip5_raw() (auto-sourced). Do not add library() calls inside the helper.
