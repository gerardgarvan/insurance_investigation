```r
# --- post-write validation ---------------------------------------------------
eps_check <- readRDS(treatment_episodes_path)
req_cols  <- c("PATID", "episode_start", "episode_end", "is_first_line")
missing_c <- setdiff(req_cols, names(eps_check))
if (length(missing_c) > 0)
  stop("[R/29] treatment_episodes.rds missing columns: ", paste(missing_c, collapse = ", "))

dup_keys <- eps_check |>
  dplyr::count(PATID, episode_start) |>
  dplyr::filter(n > 1)
if (nrow(dup_keys) > 0)
  stop("[R/29] treatment_episodes.rds has ", nrow(dup_keys), " duplicate (PATID, episode_start) rows")

if (!is.logical(eps_check$is_first_line))
  stop("[R/29] is_first_line is not logical — found: ", class(eps_check$is_first_line))

message("[R/29] treatment_episodes.rds validation: PASS — ", nrow(eps_check), " rows, ",
        dplyr::n_distinct(eps_check$PATID), " patients")
rm(eps_check, req_cols, missing_c, dup_keys)
# ---------------------------------------------------------------------------
```
