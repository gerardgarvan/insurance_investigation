```r
backup_path <- paste0(here::here("R", "00_config.R"), ".", format(Sys.time(), "%Y%m%d_%H%M%S"), ".bak")
file.copy(here::here("R", "00_config.R"), backup_path, overwrite = FALSE)

writeLines(<new_config_content>, tmp)

parse_ok <- tryCatch({ parse(tmp); TRUE }, error = function(e) { FALSE })
if (!parse_ok) {
  message("[R/98] Config parse failed — rolling back from backup")
  file.copy(backup_path, here::here("R", "00_config.R"), overwrite = TRUE)
  stop("[R/98] Config rewrite aborted: parse error in generated config.")
}

source_ok <- tryCatch({
  e <- new.env(parent = globalenv())
  source(tmp, local = e)
  TRUE
}, error = function(e) { FALSE })
if (!source_ok) {
  message("[R/98] Config source failed — rolling back from backup")
  file.copy(backup_path, here::here("R", "00_config.R"), overwrite = TRUE)
  stop("[R/98] Config rewrite aborted: source error in generated config.")
}

file.copy(tmp, here::here("R", "00_config.R"), overwrite = TRUE)
message("[R/98] 00_config.R updated successfully.")
```

### Task 3 — .bak retention cleanup (last 5 per writer)

Add a shared helper to `R/utils/utils_config_safety.R` (create if absent):

```r
# Keep the last N .bak files for a given config path; delete older ones.
# Called after a successful config write.
trim_config_bak <- function(config_path, keep = 5L) {
  dir  <- dirname(config_path)
  base <- basename(config_path)
  baks <- sort(list.files(dir, pattern = paste0("^", base, "\\.\\d{8}_\\d{6}\\.bak$"),
                           full.names = TRUE), decreasing = TRUE)
  if (length(baks) > keep) {
    old <- baks[(keep + 1):length(baks)]
    file.remove(old)
    message("[config_safety] Removed ", length(old), " old .bak file(s) for ", base)
  }
  invisible(NULL)
}
```

Add a call to `trim_config_bak(here::here("R", "00_config.R"))` at the end of the success path
in R/98 and in R/21, R/22, R/50 (after their `file.remove(backup_path)` → replace with
`trim_config_bak(...)` so they keep 5 instead of zero).
