# Plan 197-08 — R/183 MICE Diagnostic

## Objective

The 2026-11-11 MICE run produced 400 loggedEvents per model from `dist_to_source_facility_mi`,
most attributed to the `main_source` level "UCI" being dropped each iteration.

This plan:
1. Adds a `COLLAPSE_RARE_SOURCE` flag to R/183 (default OFF) that collapses `main_source` levels
   with n < 30 into "Other" before mice().
2. Writes a standalone diagnostic script `R/tools/mice_convergence_diagnostic.R` that
   reads `mice_input_patient_*.rds` and `mice_mids_*.rds` and produces the R-hat convergence table.
3. Writes the MICE-CONVERGENCE.md template (to be populated by Gerard running the diagnostic on
   HiPerGator with the existing 2026-11-11 mids file).

Do NOT rerun MICE. Do NOT change `maxit`.

### Task 2 — Write mice_convergence_diagnostic.R

Create `R/tools/mice_convergence_diagnostic.R`:

```r
# R/tools/mice_convergence_diagnostic.R
# Reads existing mice_input_patient_*.rds and mice_mids_*.rds.
# Reports: n per main_source level; Gelman-Rubin-style R-hat by iteration.
# Writes: .planning/phases/197-phase-b-pipeline-hardening/197-MICE-CONVERGENCE.md
# Run: source("R/tools/mice_convergence_diagnostic.R", local = new.env(parent = globalenv()))

suppressPackageStartupMessages({
  library(mice)
  library(dplyr)
  library(here)
})

MICE_INPUT_PATH <- Sys.getenv(
  "MICE_DIAG_INPUT",
  Sys.getenv("R183_MICE_INPUT_RDS",
    list.files(file.path(here(), "output", "internal"),
               pattern = "^mice_input_patient_.*\\.rds$", full.names = TRUE) |>
      (\(f) if (length(f) == 0) stop("No mice_input_patient_*.rds in output/internal/") else f[length(f)])()
  )
)
MICE_MIDS_PATH <- Sys.getenv(
  "MICE_DIAG_MIDS",
  Sys.getenv("R183_MICE_MIDS_RDS",
    list.files(file.path(here(), "output", "internal"),
               pattern = "^mice_mids_.*\\.rds$", full.names = TRUE) |>
      (\(f) if (length(f) == 0) stop("No mice_mids_*.rds in output/internal/") else f[length(f)])()
  )
)

OUT_MD <- file.path(here(), ".planning", "phases",
                    "197-phase-b-pipeline-hardening", "197-MICE-CONVERGENCE.md")

message("[diag] mice_input: ", MICE_INPUT_PATH)
message("[diag] mice_mids:  ", MICE_MIDS_PATH)

# --- 1. main_source level counts -------------------------------------------
mice_input <- readRDS(MICE_INPUT_PATH)
src_tbl <- mice_input |>
  count(main_source, name = "n") |>
  arrange(desc(n)) |>
  mutate(flag = if_else(n < 30, "< 30 (collapse candidate)", ""))

message("[diag] main_source level counts:")
print(src_tbl, n = Inf)

# --- 2. R-hat convergence by iteration -------------------------------------
mids <- readRDS(MICE_MIDS_PATH)
# Extract chainMean: list[variable][chain, iteration]
cm <- mids$chainMean
vars_imp <- names(cm)

rhat_rows <- lapply(vars_imp, function(v) {
  mat <- cm[[v]]  # nrow=chains (m), ncol=maxit
  if (is.null(mat) || ncol(mat) < 2) return(NULL)
  rhats <- sapply(seq_len(ncol(mat)), function(iter) {
    chains_iter <- mat[, 1:iter, drop = FALSE]
    if (iter < 2) return(NA_real_)
    B <- apply(chains_iter, 1, mean)   # chain means up to iter
    W <- apply(chains_iter, 1, var)    # within-chain var
    n <- iter
    m <- nrow(mat)
    var_hat <- ((n - 1) / n) * mean(W) + (1 / n) * var(B) * m
    rhat <- sqrt(var_hat / mean(W))
    rhat
  })
  data.frame(variable = v, iteration = seq_along(rhats), rhat = rhats)
})
rhat_df <- bind_rows(Filter(Negate(is.null), rhat_rows))

# First iteration where rhat < 1.1 and stays below
converge_iter <- rhat_df |>
  group_by(variable) |>
  summarise(
    max_rhat = max(rhat, na.rm = TRUE),
    converge_at = {
      below <- which(rhat < 1.1)
      if (length(below) == 0) NA_integer_
      else {
        # Find first run of consecutive below-1.1 that persists to end
        last_above <- max(which(rhat >= 1.1), 0L)
        if (last_above == 0L) min(below)
        else {
          after <- below[below > last_above]
          if (length(after) == 0) NA_integer_ else min(after)
        }
      }
    },
    .groups = "drop"
  )

message("[diag] Convergence table:")
print(converge_iter, n = Inf)

# --- 3. Write markdown report -----------------------------------------------
src_md <- paste0(
  "| main_source | n | flag |\n|---|---|---|\n",
  paste(sprintf("| %s | %d | %s |", src_tbl$main_source, src_tbl$n, src_tbl$flag),
        collapse = "\n")
)

conv_md <- paste0(
  "| Variable | First iter R-hat < 1.1 (stable) | Max R-hat |\n|---|---|---|\n",
  paste(sprintf("| %s | %s | %.3f |",
                converge_iter$variable,
                ifelse(is.na(converge_iter$converge_at), "not converged",
                       as.character(converge_iter$converge_at)),
                converge_iter$max_rhat),
        collapse = "\n")
)

report <- paste0(
  "# 197-MICE-CONVERGENCE\n\n",
  "Generated by R/tools/mice_convergence_diagnostic.R\n\n",
  "Input: ", basename(MICE_INPUT_PATH), "\n",
  "Mids:  ", basename(MICE_MIDS_PATH), "\n\n",
  "## main_source Level Counts\n\n", src_md, "\n\n",
  "**Collapse rule proposed:** levels with n < 30 → 'Other' (flag COLLAPSE_RARE_SOURCE=TRUE)\n",
  "**Pending D-197-NN:** Gerard's decision required before enabling.\n\n",
  "## Convergence (Gelman-Rubin-style R-hat by Iteration)\n\n", conv_md, "\n\n",
  "**Recommendation:** maxit stays at current value. ",
  "If any variable shows 'not converged', increase maxit in a future phase.\n"
)

writeLines(report, OUT_MD)
message("[diag] Report written: ", OUT_MD)
```
