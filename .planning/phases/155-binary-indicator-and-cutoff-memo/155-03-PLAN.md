```r
# ---- F_sensitivity_cutoffs sheet (Phase 155, always written) ----
# Sweeps every value in CONFIG$distance_candidate_cutoffs_mi.
# Uses distance_patient (in memory) for patient-level counts via share_ge_<c> columns.
# Written unconditionally — no dependency on distance_cutoff_mi being set.
# Per-payer wide columns satisfy D-07 (CONTEXT.md).

# Determine all payer_category_9 levels present in computed_rows
payer_levels_f <- sort(unique(computed_rows$payer_category_9))

F_sensitivity_cutoffs <- do.call(dplyr::bind_rows, lapply(cutoffs_mi, function(cm) {
  # Encounter-level: count rows in computed_rows where distance_mi >= cm
  n_enc_total   <- nrow(computed_rows)
  n_enc_flagged <- sum(computed_rows$distance_mi >= cm, na.rm = TRUE)

  # Patient-level: distance_patient has share_ge_<cm> = mean(distance_mi >= cm)
  # A patient is flagged if share_ge_<cm> > 0 (i.e., at least one encounter >= cm)
  share_col <- paste0("share_ge_", cm)
  if (share_col %in% names(distance_patient)) {
    n_pat_total   <- nrow(distance_patient)
    n_pat_flagged <- sum(distance_patient[[share_col]] > 0, na.rm = TRUE)
  } else {
    # Fallback: recompute from computed_rows if the column name doesn't match
    n_pat_total   <- dplyr::n_distinct(computed_rows$ID)
    n_pat_flagged <- computed_rows %>%
      dplyr::group_by(ID) %>%
      dplyr::summarise(any_flagged = any(distance_mi >= cm), .groups = "drop") %>%
      dplyr::pull(any_flagged) %>%
      sum(na.rm = TRUE)
  }

  # Per-payer wide columns: pct_enc_flagged_<payer> for each payer_category_9 value.
  # Column names: lowercase, spaces replaced with underscores (safe R column names).
  payer_pcts <- computed_rows %>%
    dplyr::mutate(flagged = distance_mi >= cm) %>%
    dplyr::group_by(payer_category_9) %>%
    dplyr::summarise(
      pct_enc_flagged_payer = round(100 * mean(flagged, na.rm = TRUE), 1),
      .groups = "drop"
    )

  # Build a named list of per-payer pct values and convert to wide tibble row
  payer_wide <- stats::setNames(
    as.list(
      payer_pcts$pct_enc_flagged_payer[match(payer_levels_f, payer_pcts$payer_category_9)]
    ),
    paste0("pct_enc_flagged_", gsub(" ", "_", tolower(payer_levels_f)))
  )

  dplyr::bind_cols(
    tibble::tibble(
      cutoff_mi       = cm,
      n_enc_flagged   = n_enc_flagged,
      pct_enc_flagged = round(100 * n_enc_flagged / n_enc_total, 1),
      n_pat_flagged   = n_pat_flagged,
      pct_pat_flagged = round(100 * n_pat_flagged / n_pat_total, 1)
    ),
    tibble::as_tibble(payer_wide)
  )
}))

message(glue(
  "  F_sensitivity_cutoffs: {nrow(F_sensitivity_cutoffs)} rows ",
  "(cutoffs: {paste(cutoffs_mi, collapse = ', ')} mi); ",
  "{ncol(F_sensitivity_cutoffs) - 5} per-payer columns"
))

add_styled_sheet(
  wb, "F_sensitivity_cutoffs",
  "F: Sensitivity Cutoffs — far_from_care at Candidate Thresholds",
  glue(
    "Sweeps CONFIG$distance_candidate_cutoffs_mi = c({paste(cutoffs_mi, collapse = ', ')}) mi. ",
    "n_enc_flagged and n_pat_flagged count computed-distance rows/patients with distance_mi >= cutoff_mi. ",
    "Patient flagged if any encounter meets the threshold. ",
    "pct_enc_flagged_<payer> columns show encounter-flagging rate per payer_category_9. ",
    "Always written; see E_binary_by_cutoff for the live cutoff."
  ),
  F_sensitivity_cutoffs
)
```
