```r
# Only emitted when CONFIG$distance_cutoff_mi is set (non-NA).
# Byte-identical to Phase 154 output when distance_cutoff_mi = NA.
if (!is.null(CONFIG$distance_cutoff_mi) && !is.na(CONFIG$distance_cutoff_mi)) {
  cutoff_val <- as.numeric(CONFIG$distance_cutoff_mi)
  stopifnot(
    "distance_cutoff_mi must be a single finite positive numeric" =
      length(cutoff_val) == 1L && is.finite(cutoff_val) && cutoff_val > 0
  )
  enc_distance <- enc_distance %>%
    dplyr::mutate(
      far_from_care          = dplyr::if_else(
        distance_status == "computed",
        as.integer(distance_mi >= cutoff_val),
        NA_integer_
      ),
      far_from_care_cutoff_mi = cutoff_val
    )
  message(glue(
    "  far_from_care indicator added: cutoff = {cutoff_val} mi; ",
    "{sum(enc_distance$far_from_care == 1L, na.rm = TRUE)} encounters flagged ",
    "({round(100 * mean(enc_distance$far_from_care == 1L, na.rm = TRUE), 1)}%)"
  ))
}
```

```r
# ---- E_binary_by_cutoff sheet (Phase 155, written only when cutoff is set) ----
if (!is.null(CONFIG$distance_cutoff_mi) && !is.na(CONFIG$distance_cutoff_mi)) {
  cutoff_val_e <- as.numeric(CONFIG$distance_cutoff_mi)

  # Encounter-level flagging by payer_category_9
  e_enc <- computed_rows %>%
    dplyr::mutate(far_from_care = as.integer(distance_mi >= cutoff_val_e)) %>%
    dplyr::group_by(payer_category_9) %>%
    dplyr::summarise(
      n_total  = dplyr::n(),
      n_flagged = sum(far_from_care, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    dplyr::bind_rows(
      dplyr::summarise(.,
        payer_category_9 = "Overall",
        n_total  = sum(n_total),
        n_flagged = sum(n_flagged)
      )
    ) %>%
    dplyr::mutate(
      grain      = "encounter",
      pct_flagged = round(100 * n_flagged / n_total, 1)
    ) %>%
    dplyr::select(grain, payer_category_9, n_total, n_flagged, pct_flagged)

  # Patient-level flagging: a patient is flagged if ANY of their computed encounters >= cutoff
  e_pat_base <- computed_rows %>%
    dplyr::mutate(far_from_care = as.integer(distance_mi >= cutoff_val_e)) %>%
    dplyr::group_by(ID, payer_category_9) %>%
    dplyr::summarise(
      flagged = any(far_from_care == 1L),
      .groups = "drop"
    )

  e_pat <- e_pat_base %>%
    dplyr::group_by(payer_category_9) %>%
    dplyr::summarise(
      n_total   = dplyr::n(),
      n_flagged = sum(flagged),
      .groups   = "drop"
    ) %>%
    dplyr::bind_rows(
      dplyr::summarise(.,
        payer_category_9 = "Overall",
        n_total   = sum(n_total),
        n_flagged = sum(n_flagged)
      )
    ) %>%
    dplyr::mutate(
      grain       = "patient",
      pct_flagged = round(100 * n_flagged / n_total, 1)
    ) %>%
    dplyr::select(grain, payer_category_9, n_total, n_flagged, pct_flagged)

  E_binary_by_cutoff <- dplyr::bind_rows(e_enc, e_pat)

  add_styled_sheet(
    wb, "E_binary_by_cutoff",
    glue("E: Binary Indicator — Cutoff {cutoff_val_e} mi"),
    glue("far_from_care = 1 when distance_mi >= {cutoff_val_e}. Encounter-level rows then patient-level rows. Payer: payer_category_9 (9-category canonical). Regenerate after updating CONFIG$distance_cutoff_mi."),
    E_binary_by_cutoff
  )

  message(glue("  E_binary_by_cutoff sheet added (cutoff = {cutoff_val_e} mi)"))
}
```
