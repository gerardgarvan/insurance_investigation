#' Load and validate the surveillance strategy codeset
#'
#' Reads surveillance_codeset.xlsx and validates structural rules per D-20/21/22.
#' All code columns forced to character. code_norm is trimmed, uppercased,
#' dots stripped.
#'
#' @param path Character. Path to xlsx file.
#' @return Validated tibble with columns: modality, submodality, code,
#'   code_norm, cdm_table, type_filter, match, tier.
load_surveillance_codeset <- function(
    path = here::here("data/reference/surveillance_codeset.xlsx")) {

  required_cols <- c("modality", "submodality", "code", "code_norm",
                     "cdm_table", "type_filter", "match", "tier")
  valid_tiers   <- c("primary", "sensitivity")
  valid_matches <- c("exact", "prefix", "component_all_same_day")

  cs <- readxl::read_excel(path, sheet = "Analysis_Codeset",
                            col_types = "text")

  missing_cols <- setdiff(required_cols, names(cs))
  if (length(missing_cols) > 0)
    stop(glue::glue("surveillance_codeset missing required columns: ",
                    paste(missing_cols, collapse = ", ")))

  bad_tiers <- setdiff(unique(cs$tier), valid_tiers)
  if (length(bad_tiers) > 0)
    stop(glue::glue("Invalid tier values: {paste(bad_tiers, collapse=', ')}. ",
                    "Must be one of: {paste(valid_tiers, collapse=', ')}"))

  bad_matches <- setdiff(unique(cs$match), valid_matches)
  if (length(bad_matches) > 0)
    stop(glue::glue("Invalid match values: {paste(bad_matches, collapse=', ')}. ",
                    "Must be one of: {paste(valid_matches, collapse=', ')}"))

  # Normalize code_norm: trim, uppercase, strip dots
  cs <- cs %>%
    dplyr::mutate(
      code_norm = stringr::str_trim(code_norm) %>%
                  toupper() %>%
                  stringr::str_remove_all("\\.")
    )

  # Duplicate check: modality x code_norm
  dups <- cs %>%
    dplyr::count(modality, code_norm) %>%
    dplyr::filter(n > 1)
  if (nrow(dups) > 0)
    stop(glue::glue("Duplicate modality x code_norm rows: ",
                    paste(paste(dups$modality, dups$code_norm, sep="/"),
                          collapse="; ")))

  # component_all_same_day validation
  comp_rows <- cs %>% dplyr::filter(match == "component_all_same_day")
  if (nrow(comp_rows) > 0) {
    bad_comp <- comp_rows %>%
      dplyr::mutate(n_comp = lengths(strsplit(code_norm, ";"))) %>%
      dplyr::filter(n_comp < 2 | cdm_table != "LAB_RESULT_CM")
    if (nrow(bad_comp) > 0)
      stop(paste("component_all_same_day rows must have >=2 semicolon-separated",
                 "components and cdm_table=LAB_RESULT_CM"))
  }

  message(glue::glue("  Codeset loaded: {nrow(cs)} rows, ",
                     "{dplyr::n_distinct(cs$modality)} modalities"))
  cs
}

# Internal helper: apply HL denominator logic to an in-memory tibble.
# Accepts tibble(ID, DX, DX_TYPE, DX_DATE). Returns tibble(ID, hl_anchor_date).
# Used by get_hl_any_dx_ids() and by testthat fixtures (no DuckDB needed).
.hl_any_dx_from_tibble <- function(dx_tbl) {
  dx_tbl %>%
    dplyr::filter(
      (DX_TYPE == "10" &
         stringr::str_starts(
           stringr::str_remove_all(stringr::str_trim(toupper(DX)), "\\."),
           "C81")) |
      (DX_TYPE == "09" &
         stringr::str_starts(
           stringr::str_remove_all(stringr::str_trim(DX), "\\."),
           "201"))
    ) %>%
    dplyr::group_by(ID) %>%
    dplyr::summarise(hl_anchor_date = min(DX_DATE, na.rm = TRUE),
                     .groups = "drop")
}

#' Get all patients with at least one HL diagnosis code
#'
#' Returns tibble with HL any-dx denominator: ICD-10 C81* (DX_TYPE="10") or
#' ICD-9 201* (DX_TYPE="09"). Anchor date = min(DX_DATE) per patient.
#' in_confirmed_cohort = ID %in% get_hl_patient_ids().
#' Uses DuckDB prefix pushdown before collect(). (D-04)
#'
#' @return Tibble: ID (chr), hl_anchor_date (Date), in_confirmed_cohort (lgl).
#'   Zero-row tibble with correct types on error.
get_hl_any_dx_ids <- function() {
  dx_tbl <- safe_table("DIAGNOSIS")
  if (is.null(dx_tbl)) {
    message("  Warning: DIAGNOSIS table not found; returning empty tibble")
    return(tibble::tibble(ID = character(),
                          hl_anchor_date = as.Date(NA_character_),
                          in_confirmed_cohort = logical()))
  }

  tryCatch({
    # DuckDB prefix pushdown: materialize only C81* and 201* rows
    raw <- dx_tbl %>%
      dplyr::filter(
        (DX_TYPE == "10" &
           (stringr::str_starts(
              toupper(stringr::str_remove_all(stringr::str_trim(DX), "\\.")),
              "C81"))) |
        (DX_TYPE == "09" &
           stringr::str_starts(
              stringr::str_remove_all(stringr::str_trim(DX), "\\."),
              "201"))
      ) %>%
      dplyr::select(ID, DX_DATE) %>%
      dplyr::collect()

    confirmed_ids <- get_hl_patient_ids()

    result <- raw %>%
      dplyr::group_by(ID) %>%
      dplyr::summarise(
        hl_anchor_date = min(DX_DATE, na.rm = TRUE),
        .groups = "drop"
      ) %>%
      dplyr::mutate(
        hl_anchor_date      = as.Date(hl_anchor_date),
        in_confirmed_cohort = ID %in% confirmed_ids
      )

    # Guard: every confirmed ID must appear in any-dx set
    missing_confirmed <- setdiff(confirmed_ids, result$ID)
    if (length(missing_confirmed) > 0)
      warning(glue::glue("{length(missing_confirmed)} confirmed-cohort IDs ",
                         "NOT found in HL any-dx denominator. Check ICD_CODES."))

    message(glue::glue(
      "  HL any-dx denominator: {format(nrow(result), big.mark=',')} patients ",
      "({sum(result$in_confirmed_cohort)} in confirmed cohort)"
    ))
    result

  }, error = function(e) {
    message(glue::glue("  Error in get_hl_any_dx_ids(): {e$message}"))
    tibble::tibble(ID = character(),
                   hl_anchor_date = as.Date(NA_character_),
                   in_confirmed_cohort = logical())
  })
}
