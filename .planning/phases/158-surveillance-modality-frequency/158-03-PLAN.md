```r
message("SECTION N: Phase 158 surveillance codeset checks")

if (!exists("load_surveillance_codeset")) source("R/utils/utils_treatment.R")

cs_check <- load_surveillance_codeset()
stopifnot(
  "Codeset must have 108 rows (105 baseline + 3 stress-echo D-20)" =
    nrow(cs_check) == 108,
  "submodality column must be present (D-21)" =
    "submodality" %in% names(cs_check),
  "All match values must be in allowed set" =
    all(cs_check$match %in% c("exact","prefix","component_all_same_day")),
  "Thyroid rows must be renamed from 'Thyroid stimulating hormone'" =
    !any(cs_check$modality == "Thyroid stimulating hormone"),
  "Stress test rows for 93350/93351/93352 must exist (D-20)" =
    sum(cs_check$modality == "Stress test" &
          trimws(cs_check$code_norm) %in% c("93350","93351","93352")) == 3,
  "CBC component_all_same_day row must exist (D-22)" =
    any(cs_check$match == "component_all_same_day"),
  "get_hl_any_dx_ids must be defined (158-01)" =
    exists("get_hl_any_dx_ids")
)
message("  PASS: surveillance codeset and function checks")
```
