# Bind all raw events; ensure common columns exist before bind
bind_safe <- function(...) {
  lst <- list(...) %>% keep(~ nrow(.) > 0)
  if (length(lst) == 0) return(tibble(ID=character(), code_norm=character(),
                                       modality=character(), tier=character(),
                                       source_table=character(), event_date=as.Date(NA)))
  bind_rows(lst)
}

  events_proc_cpt_exact, events_proc_cpt_prefix,
  events_proc_icd10, events_proc_icd9, events_proc_icd10_prefix,
  events_lab_exact, events_lab_prefix,
  events_cbc,
  events_diag_exact, events_diag_prefix

  filter(!is.na(event_date))

    present             = n_records > 0
  )

            nrow(A_code_presence) == nrow(codeset))

compute_modality_stats <- function(events_subset, denom_df) {
  denom_n <- nrow(denom_df)
  events_subset %>%
    group_by(modality) %>%
    summarise(
      n_patients         = n_distinct(ID),
      pct_denominator    = n_distinct(ID) / denom_n * 100,
      events_per_pt_median = median(as.numeric(table(ID))),
      events_per_pt_iqr    = IQR(as.numeric(table(ID))),
      events_per_pt_max    = max(as.numeric(table(ID))),
      .groups = "drop"
    ) %>%
    left_join(
      # events per person-year (post-anchor only)
      events_subset %>%
        filter(pre_post_anchor == "post_anchor") %>%
        count(ID, modality, name = "n_events") %>%
        left_join(denom_df %>% select(ID, person_years), by = "ID") %>%
        group_by(modality) %>%
        summarise(
          events_per_py = sum(n_events, na.rm=TRUE) /
                          sum(person_years, na.rm=TRUE),
          .groups = "drop"
        ),
      by = "modality"
    )
}
