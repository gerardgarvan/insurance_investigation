  <behavior>
Returns the count of cohort patients PRESENT in addr_raw whose address rows carry no usable raw ADDRESS_ZIP5 -- the pre-coalesce baseline. Restricted to present patients so it is directly comparable to n_present_no_usable_zip5; absent patients are C-02c's business, not this function's.

R code:

  # 140-09-PATCH FIX-17: pre-coalesce baseline -- raw ADDRESS_ZIP5 only, no ZIP9-derived
  # fallback. normalize_zip5_raw() is called, not modified (Out of Scope, CONTEXT Section 11).
  compute_c02_baseline <- function(cohort_ids, addr_raw) {
    base_tbl <- addr_raw %>%
      mutate(.zip5_raw_only = normalize_zip5_raw(ADDRESS_ZIP5)) %>%
      group_by(ID) %>%
      summarise(has_any_zip5 = any(!is.na(.zip5_raw_only)), .groups = "drop")

    present_ids <- intersect(cohort_ids, addr_raw$ID)

    tibble(ID = present_ids) %>%
      left_join(base_tbl, by = "ID") %>%
      mutate(has_any_zip5 = coalesce(has_any_zip5, FALSE)) %>%
      summarise(n = sum(!has_any_zip5)) %>%
      pull(n)
  }
