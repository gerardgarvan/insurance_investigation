    ```r
    run_gee <- function(data, outcome, exposure, label) {
      d <- data |>
        dplyr::filter(!is.na(.data[[exposure]])) |>
        dplyr::arrange(ID, year_index) |>
        dplyr::mutate(year_index_cat = droplevels(year_index_cat),
                      y = .data[[outcome]], x = .data[[exposure]])
      warns <- character(0)
      fit <- tryCatch(
        withCallingHandlers(
          geepack::geeglm(y ~ x + year_index_cat, data = d, id = ID,
                          family = binomial("logit"), corstr = GEE_CORSTR),
          warning = function(w) { warns <<- c(warns, conditionMessage(w)); invokeRestart("muffleWarning") }),
        error = function(e) e)
      base <- list(label = label, n_py = nrow(d), n_pat = dplyr::n_distinct(d$ID),
                   warnings = paste(unique(warns), collapse = " | "))
      if (inherits(fit, "error")) {
        nb <- naive_or_se(matrix(as.numeric(table(factor(d$x, 0:1), factor(d$y, 0:1))), 2))
        return(c(base, list(method = "CRUDE FALLBACK (GEE error)", or = exp(nb$log_or),
                            ci_lo = NA_real_, ci_hi = NA_real_, p = NA_real_, error = conditionMessage(fit))))
      }
      ct <- summary(fit)$coefficients
      b <- ct["x", "Estimate"]; se <- ct["x", "Std.err"]          # geepack robust SE
      c(base, list(method = glue::glue("GEE ({GEE_CORSTR})"), or = exp(b),
                   ci_lo = exp(b - 1.96 * se), ci_hi = exp(b + 1.96 * se),
                   p = ct["x", "Pr(>|W|)"], error = NA_character_))
    }
    ```
