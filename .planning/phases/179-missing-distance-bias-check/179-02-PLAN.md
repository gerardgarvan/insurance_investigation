**Section 5 — GEE model (D-11, D-12):**

Wrap each geeglm call in tryCatch; on error, store the error message and report crude pooled OR
as fallback (D-13 requirement: "do not switch method silently").

```r
run_gee <- function(data, label) {
  tryCatch({
    fit <- geepack::geeglm(
      cbc_year ~ far + year_index_cat,
      data   = data,
      id     = factor(data$ID),
      family = binomial(link = "logit"),
      corstr = GEE_CORSTR
    )
    coef_tbl <- summary(fit)$coefficients
    log_or  <- coef_tbl["far", "Estimate"]
    se_log  <- coef_tbl["far", "Std. Error"]
    list(
      label   = label,
      or      = exp(log_or),
      ci_lo   = exp(log_or - 1.96 * se_log),
      ci_hi   = exp(log_or + 1.96 * se_log),
      p       = coef_tbl["far", "Pr(>|W|)"],
      n_py    = nrow(data),
      n_pat   = dplyr::n_distinct(data$ID),
      corstr  = GEE_CORSTR,
      converged = fit$geese$error == 0L,
      error   = NA_character_,
      fallback_or = NA_real_
    )
  }, error = function(e) {
    message(glue("  GEE failed ({label}): {conditionMessage(e)}"))
    # Fallback crude pooled OR (D-13)
    ct <- table(far = data$far, cbc_year = data$cbc_year)
    nb <- naive_or_se(as.matrix(ct))
    list(
      label   = label, or = exp(nb$log_or), ci_lo = NA_real_, ci_hi = NA_real_,
      p = NA_real_, n_py = nrow(data), n_pat = dplyr::n_distinct(data$ID),
      corstr = GEE_CORSTR, converged = FALSE,
      error = conditionMessage(e), fallback_or = exp(nb$log_or)
    )
  })
}

gee_primary   <- run_gee(py_primary, "Primary (years 2+, patient-level median exposure)")
gee_secondary <- run_gee(py_all,     "Secondary (all years, patient-level median exposure)")
```
