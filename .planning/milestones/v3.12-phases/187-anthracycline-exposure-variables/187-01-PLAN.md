### SECTION 2 — Compute Flags and Time Variables

Put the whole computation in ONE function so the synthetic fixtures and the real data go through the
identical code path (D-16). No separate copy of the logic for the fixtures.

```r
build_patient_out <- function(episodes, followup, drugs) {
  # episodes: ID (chr), treatment_type, drug_name, admin_date (Date), first_line, episode_id
  # followup: ID (chr), follow_end (Date) — one row per cohort patient
  chemo_episodes <- dplyr::filter(episodes, treatment_type == "Chemotherapy")      # D-04b

  chemo_ids <- dplyr::distinct(chemo_episodes, ID)

  dose_tbl <- last_anthracycline_dose(chemo_episodes, drugs)                        # D-14
  # dose_tbl columns: ID, last_dose_firstline_dt, last_dose_ever_dt -> use last_dose_ever_dt

  drugs_received_tbl <- chemo_episodes |>
    dplyr::filter(drug_name %in% drugs) |>
    dplyr::group_by(ID) |>
    dplyr::summarise(drugs_received = paste(sort(unique(drug_name)), collapse = "; "),
                     .groups = "drop")                                              # D-05

  out <- followup |>
    dplyr::mutate(any_chemo = ID %in% chemo_ids$ID) |>
    dplyr::left_join(dplyr::select(dose_tbl, ID, last_anthracycline_date = last_dose_ever_dt),
                     by = "ID") |>
    dplyr::left_join(drugs_received_tbl, by = "ID") |>
    dplyr::mutate(
      # Flag from presence of an anthracycline row (not from the date), so an NA admin_date
      # can never silently turn an exposed patient into FALSE.
      any_anthracycline = !is.na(drugs_received),                                   # D-06, never NA
      dose_after_end    = any_anthracycline & !is.na(follow_end) &
                          last_anthracycline_date > follow_end,                     # D-07 flag
      years_since_last_anthracycline = dplyr::if_else(
        any_anthracycline & !is.na(follow_end) & !dose_after_end,
        as.numeric(follow_end - last_anthracycline_date) / 365.25,
        NA_real_
      )
    )

  # An exposed patient must have a date; a mismatch means an anthracycline row had NA admin_date.
  bad <- out$any_anthracycline & is.na(out$last_anthracycline_date)
  if (any(bad)) stop("[187] ", sum(bad), " patient(s) have an anthracycline row but NA last date — check admin_date NAs.")

  out
}
```

```r
sup <- function(n) {
  dplyr::case_when(
    n == 0L            ~ "0",
    n >= 1L & n <= 10L ~ "<11",
    TRUE               ~ as.character(n)
  )
}
sup_pair <- function(n_true, n_false) {
  s <- c(sup(n_true), sup(n_false))
  if (any(s == "<11")) s <- c("<11", "<11")
  s
}

n_total <- nrow(patient_out)
```
